   146→export function getScoreColor(score: number | null): string {
   147→  if (score === null) return 'bg-gray-100 text-gray-500 dark:bg-gray-800 dark:text-gray-400';
   148→  if (score >= 80) return 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200';
   149→  if (score >= 50) return 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-200';
   150→  return 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200';
   151→}
   152→
   153→export function getScoreBadgeClass(score: number | null): string {
   154→  if (score === null) return 'border-gray-300 text-gray-500';
   155→  if (score >= 80) return 'border-green-500 text-green-700';
   156→  if (score >= 50) return 'border-yellow-500 text-yellow-700';
   157→  return 'border-red-500 text-red-700';
   158→}
   159→
   160→export function getAuditColor(type: AuditOpinionType): string {
   161→  switch (type) {
   162→    case 'Unmodified': return 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200';
   163→    case 'Qualified': return 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-200';
   164→    case 'Adverse': return 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200';
   165→    case 'Disclaimer': return 'bg-red-200 text-red-900 dark:bg-red-900 dark:text-red-100';
   166→    default: return 'bg-gray-100 text-gray-500';
   167→  }
   168→}
   169→
   170→export function getCoalitionColor(coalition: CoalitionType): string {
   171→  switch (coalition) {
   172→    case 'Kenya Kwanza': return 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200 border-green-300';
   173→    case 'Azimio': return 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-200 border-blue-300';
   174→    case 'Independent': return 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-300 border-gray-300';
   175→    case 'Other': return 'bg-purple-100 text-purple-800 dark:bg-purple-900 dark:text-purple-200 border-purple-300';
   176→    default: return 'bg-gray-100 text-gray-500';
   177→  }
   178→}
   179→
   180→export function getRegionForCounty(countyName: string): RegionType {
   181→  for (const [region, counties] of Object.entries(REGIONS)) {
   182→    if (counties.includes(countyName)) return region as RegionType;
   183→  }
   184→  return 'Rift Valley';
   185→}

   668→// Helper: attach county-level leadership officials (DG, Senator, Woman Rep) from a
   669→// county-specific extras object. Used by Nairobi, Mombasa, Kisumu, Nakuru, Kiambu,
   670→// Kakamega, Meru, Machakos — and extensible to other counties as data is verified.
   671→interface CountyExtraOfficial {
   672→  fullName: string;
   673→  party: CoalitionType | string;
   674→  coalition: CoalitionType;
   675→  biography: string;
   676→  biographySource: string;
   677→}
   678→interface CountyExtraOfficials {
   679→  deputyGovernor?: CountyExtraOfficial;
   680→  senator?: CountyExtraOfficial;
   681→  womanRep?: CountyExtraOfficial;
   682→}
   683→
   684→function attachCountyLeadership(
   685→  countyData: CountyData,
   686→  governorRep: Representative,
   687→  countyName: string,
   688→  countyCode: number,
   689→  extras: CountyExtraOfficials,
   690→): void {
   691→  const buildRep = (
   692→    role: 'dep' | 'sen' | 'wrep',
   693→    roleTitle: string,
   694→    o: CountyExtraOfficial,
   695→  ): Representative => ({
   696→    ...governorRep,
   697→    id: `${role}-${countyCode}-${countyName.toLowerCase().replace(/[^a-z0-9]/g, '-')}`,
   698→    fullName: o.fullName,
   699→    officialTitle: `${roleTitle}, ${countyName} County`,
   700→    party: o.party as string,
   701→    coalition: o.coalition,
   702→    biography: o.biography,
   703→    biographySource: o.biographySource,
   704→    scorecard: makeDefaultScorecard(),
   705→    auditOpinion: null,
   706→    budgetPerformance: makeDefaultBudgetPerformance(),
   707→  });
   708→
   709→  if (extras.deputyGovernor) {
   710→    countyData.deputyGovernor = buildRep('dep', 'Deputy Governor', extras.deputyGovernor);
   711→  }
   712→  if (extras.senator) {
   713→    countyData.senator = buildRep('sen', 'Senator', extras.senator);
   714→  }
   715→  if (extras.womanRep) {
   716→    countyData.womanRep = buildRep('wrep', 'Woman Representative', extras.womanRep);
   717→  }
   718→}

   989→export interface FilterState {
   990→  searchQuery: string;
   991→  region: RegionType | 'All';
   992→  coalition: CoalitionType | 'All';
   993→  level: LevelType | 'All';
   994→  scoreRange: [number, number];
   995→  auditOpinion: AuditOpinionType | 'All';
   996→}

   998→export const DEFAULT_FILTERS: FilterState = {
   999→  searchQuery: '',
  1000→  region: 'All',
  1001→  coalition: 'All',
  1002→  level: 'All',
  1003→  scoreRange: [0, 100],
  1004→  auditOpinion: 'All',

  1007→export function filterCounties(counties: CountyData[], filters: FilterState): CountyData[] {
  1008→  return counties.filter(county => {
  1009→    const gov = county.governor;
  1010→    // Region filter
  1011→    if (filters.region !== 'All' && county.region !== filters.region) return false;
  1012→    // Coalition filter
  1013→    if (filters.coalition !== 'All' && gov.coalition !== filters.coalition) return false;
  1014→    // Level filter
  1015→    if (filters.level !== 'All' && gov.level !== filters.level) return false;
  1016→    // Audit opinion filter
  1017→    if (filters.auditOpinion !== 'All' && gov.auditOpinion?.fy2023_24.type !== filters.auditOpinion) return false;
  1018→    // Score range filter
  1019→    const overallScore = gov.scorecard.overallAccountability.score ?? 0;
  1020→    if (overallScore < filters.scoreRange[0] || overallScore > filters.scoreRange[1]) return false;
  1021→    // Search query
  1022→    if (filters.searchQuery) {
  1023→      const q = filters.searchQuery.toLowerCase();
  1024→      const searchable = `${county.name} ${gov.fullName} ${gov.party} ${gov.coalition}`.toLowerCase();
  1025→      if (!searchable.includes(q)) return false;
  1026→    }
  1027→    return true;
  1028→  });
  1029→}

  1031→export function searchRepresentatives(query: string): Representative[] {
  1032→  const results: Representative[] = [];
  1033→  const q = query.toLowerCase();
  1034→  const counties = buildAllCountyData();
  1035→
  1036→  // Search national level
  1037→  const national = [NATIONAL_SUMMARY.president, NATIONAL_SUMMARY.deputyPresident];
  1038→  for (const rep of national) {
  1039→    const searchable = `${rep.fullName} ${rep.officialTitle} ${rep.party} ${rep.coalition} ${rep.jurisdiction}`.toLowerCase();
  1040→    if (searchable.includes(q)) results.push(rep);
  1041→  }
  1042→
  1043→  // Search all county data
  1044→  for (const county of counties) {
  1045→    const countyReps = flattenCountyRepresentatives(county);
  1046→    for (const rep of countyReps) {
  1047→      const searchable = `${rep.fullName} ${rep.officialTitle} ${rep.party} ${rep.coalition} ${rep.jurisdiction}`.toLowerCase();
  1048→      if (searchable.includes(q)) results.push(rep);
  1049→    }
  1050→  }
  1051→
  1052→  return results;
  1053→}
  1054→
  1055→export function flattenCountyRepresentatives(county: CountyData): Representative[] {
  1056→  const reps: Representative[] = [];
  1057→  reps.push(county.governor);
  1058→  if (county.deputyGovernor) reps.push(county.deputyGovernor);
  1059→  if (county.senator) reps.push(county.senator);
  1060→  if (county.womanRep) reps.push(county.womanRep);
  1061→  if (county.constituencyMPs) reps.push(...county.constituencyMPs);
  1062→  if (county.assemblySpeaker) reps.push(county.assemblySpeaker);
  1063→  if (county.deputySpeaker) reps.push(county.deputySpeaker);
  1064→  if (county.electedMCAs) reps.push(...county.electedMCAs);
  1065→  if (county.nominatedMCAs) reps.push(...county.nominatedMCAs);
  1066→  if (county.cecms) reps.push(...county.cecms);
  1067→  if (county.countySecretary) reps.push(county.countySecretary);
  1068→  if (county.countyAttorney) reps.push(county.countyAttorney);
  1069→  return reps;
  1070→}

  1081→export function getRepresentativeById(id: string): Representative | null {
  1082→  // National level
  1083→  if (NATIONAL_SUMMARY.president.id === id) return NATIONAL_SUMMARY.president;

  1088→  for (const county of counties) {

  1092→    }
  1093→  }
  1094→  return null;
